"A Promise is a JavaScript object used to handle asynchronous operations. It represents a result that may be available now, later, or may fail. It has three states: pending, fulfilled, and rejected. We handle the result using methods like .then(), .catch(), and .finally().
 Promises are commonly used for API calls, database operations, and file handling."

 const myPromise = new Promise((resolve, reject) => {
  let success = true;

  if (success) {
    resolve("Task completed successfully");
  } else {
    reject("Something went wrong");
  }
});

myPromise
  .then((result) => {
    console.log(result);
  })
  .catch((error) => {
    console.log(error);
  });
